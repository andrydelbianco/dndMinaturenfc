        const storageKey = 'dnd_save_' + characterData.name.replace(/\s+/g, '_');

 

        function loadSavedData() {

            const saved = localStorage.getItem(storageKey);

            if (saved) {

                const parsed = JSON.parse(saved);

                if (parsed.hpCurrent !== undefined) characterData.hpCurrent = parsed.hpCurrent;

                if (parsed.inventory) characterData.inventory = parsed.inventory;

                if (parsed.inspiration !== undefined) characterData.inspiration = parsed.inspiration;

                if (parsed.spellSlots) characterData.spellSlots = parsed.spellSlots;

                if (parsed.hitDiceUsed !== undefined) characterData.hitDiceUsed = parsed.hitDiceUsed;

                if (parsed.notes) document.getElementById('session-notes').value = parsed.notes;

            }

        }

            const dataToSave = {

            };

        function shortRest() {

            let maxHD = characterData.level; // Stesso numero del livello

            let available = maxHD - characterData.hitDiceUsed;

           

            if(available <= 0) {

                alert("Non hai più Dadi Vita disponibili per il riposo breve!");

                return;

            }

           

            let input = prompt(`Quanti Dadi Vita (d6) vuoi usare? (Disponibili: ${available})`, "1");

            if(input === null || input === "") return;

           

            let numDice = parseInt(input);

            if(isNaN(numDice) || numDice < 1 || numDice > available) {

                alert("Numero non valido!");

                return;

            }

 

            // Stregone: d6. Estrae Costituzione (+2 -> 2)

            let conMod = parseInt(characterData.stats.COS.replace('+', ''));

            let totalHeal = 0;

            let rollDetails = [];

 

            // Lancia i dadi

            for(let i=0; i<numDice; i++) {

                let roll = Math.floor(Math.random() * 6) + 1;

                totalHeal += roll + conMod;

                rollDetails.push(roll);

            }

 

            // Scala i dadi usati, cura il giocatore, salva.

            characterData.hitDiceUsed += numDice;

            changeHP(totalHeal);

           

            // Registra nel Log dei dadi (con un colore visibile)

            const time = new Date().toLocaleTimeString('it-IT', { hour: '2-digit', minute: '2-digit', second: '2-digit' });

            let logMsg = `

                <strong style="color: var(--secondary-color)">Riposo Breve (${numDice}d6+${numDice * conMod})</strong>

                ➔ Cura: <span style="color:var(--secondary-color); font-size:16px;"><b>${totalHeal}</b></span>

                <div style="color:#777; font-size:11px; margin-top:3px;">(Dadi estratti: ${rollDetails.join(', ')}) - ${time}</div>

            `;

            rollHistory.unshift(logMsg);

            document.getElementById('dice-log').innerHTML = rollHistory.map(log => `<li>${log}</li>`).join('');

        let rollHistory = [];

        function rollDice(sides) {

            const result = Math.floor(Math.random() * sides) + 1;

            document.getElementById('dice-result').innerText = result;

            document.getElementById('dice-label').innerText = `Risultato d${sides}`;

           

            const time = new Date().toLocaleTimeString('it-IT', { hour: '2-digit', minute: '2-digit', second: '2-digit' });

            rollHistory.unshift(`<strong>d${sides}</strong> ➔ ${result} <span style="color:#777; font-size:11px;">(${time})</span>`);

            if (rollHistory.length > 5) { rollHistory.pop(); }

           

            document.getElementById('dice-log').innerHTML = rollHistory.map(log => `<li>${log}</li>`).join('');

        }
